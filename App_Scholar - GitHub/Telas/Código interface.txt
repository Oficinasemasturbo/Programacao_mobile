// TELA DE ALUNOS
// ======================================================

function TelaAlunos({ voltar }) {

  const [alunos, setAlunos] = useState([]);
  const [nome, setNome] = useState('');
  const [cpf, setCpf] = useState('');
  const [idade, setIdade] = useState('');
  const [idEditando, setIdEditando] = useState(null);

  const [telaAluno, setTelaAluno] = useState('inicio');
  const [alunoSelecionado, setAlunoSelecionado] = useState(null);

  const API = 'http://10.121.184.57/gabriel/App_Scholar_Api';

  // funções aqui...


  // ======================================================
  // CONSULTAR ALUNOS
  // ======================================================

  async function consultarAlunos() {

  try {

    console.log('Consultando alunos...');

    const url = `${API}/consultar_alunos.php`;

    console.log('URL:', url);

    const resposta = await fetch(url);

    console.log('Status HTTP:', resposta.status);

    const texto = await resposta.text();

    console.log('Resposta do servidor:', texto);

    if (!resposta.ok) {

      throw new Error(
        `Servidor respondeu com HTTP ${resposta.status}`
      );
    }

    if (!texto.trim()) {

      throw new Error(
        'O servidor retornou uma resposta vazia.'
      );
    }

    let dados;

    try {

      dados = JSON.parse(texto);

    } catch (erroJSON) {

      console.log('Erro ao converter JSON:', erroJSON);

      throw new Error(
        'O servidor não retornou JSON válido.\n\n' +
        texto
      );
    }

    console.log('Alunos recebidos:', dados);

    if (!Array.isArray(dados)) {

      throw new Error(
        'O servidor não retornou uma lista de alunos.'
      );
    }

    setAlunos(dados);

    setTelaAluno('lista');

  } catch (erro) {

    console.log('ERRO AO CONSULTAR:', erro);

    Alert.alert(
      'Erro ao consultar alunos',
      erro.message
    );
  }
}



  // ======================================================
  // INSERIR ALUNO
  // ======================================================

  async function inserirAluno() {

    if (!nome.trim()) {

      Alert.alert(
        'Atenção',
        'Digite o nome do aluno.'
      );

      return;
    }

    try {

      const resposta = await fetch(
        `${API}/inserir_aluno.php`,
        {
          method: 'POST',

          headers: {
            'Content-Type': 'application/json',
          },

          body: JSON.stringify({
            nome: nome,
            cpf: cpf,
            idade: idade
              ? Number(idade)
              : null,
          }),
        }
      );

      const dados = await resposta.json();

      if (dados.sucesso) {

        Alert.alert(
          'Sucesso',
          'Aluno cadastrado!'
        );

        setNome('');
        setCpf('');
        setIdade('');

        consultarAlunos();

      } else {

        Alert.alert(
          'Erro',
          dados.mensagem || 'Erro ao cadastrar.'
        );
      }

    } catch (erro) {

      console.log(erro);

      Alert.alert(
        'Erro',
        'Não foi possível conectar com o servidor.'
      );
    }
  }


  // ======================================================
  // PREPARAR EDIÇÃO
  // ======================================================

  function prepararEdicao(aluno) {

    setAlunoSelecionado(aluno);

    setIdEditando(aluno.id);

    setNome(aluno.nome);

    setCpf(aluno.cpf || '');

    setIdade(
      aluno.idade
        ? String(aluno.idade)
        : ''
    );

    setTelaAluno('detalhes');
  }


  // ======================================================
  // ATUALIZAR ALUNO
  // ======================================================

  async function atualizarAluno() {

    if (!idEditando) {

      Alert.alert(
        'Atenção',
        'Nenhum aluno selecionado.'
      );

      return;
    }

    try {

      const resposta = await fetch(
        `${API}/atualizar_aluno.php`,
        {
          method: 'POST',

          headers: {
            'Content-Type': 'application/json',
          },

          body: JSON.stringify({
            id: idEditando,
            nome: nome,
            cpf: cpf,
            idade: idade
              ? Number(idade)
              : null,
          }),
        }
      );

      const dados = await resposta.json();

      if (dados.sucesso) {

        Alert.alert(
          'Sucesso',
          'Aluno atualizado!'
        );

        setNome('');
        setCpf('');
        setIdade('');
        setIdEditando(null);
        setAlunoSelecionado(null);

        consultarAlunos();

      } else {

        Alert.alert(
          'Erro',
          dados.mensagem || 'Erro ao atualizar.'
        );
      }

    } catch (erro) {

      console.log(erro);

      Alert.alert(
        'Erro',
        'Não foi possível conectar com o servidor.'
      );
    }
  }


  // ======================================================
  // INATIVAR ALUNO
  // ======================================================

  async function inativarAluno(id) {

    Alert.alert(
      'Inativar aluno',

        {
          text: 'Cancelar',
          style: 'cancel',
        },

        {
          text: 'Inativar',
          style: 'destructive',

          onPress: async () => {

            try {

              const resposta = await fetch(
                `${API}/inativar_aluno.php`,
                {
                  method: 'POST',

                  headers: {
                    'Content-Type': 'application/json',
                  },

                  body: JSON.stringify({
                    id: id,
                  }),
                }
              );

              const dados = await resposta.json();

              if (dados.sucesso) {

                Alert.alert(
                  'Sucesso',
                  'Aluno inativado!'
                );

                setAlunoSelecionado(null);

                consultarAlunos();

              } else {

                Alert.alert(
                  'Erro',
                  dados.mensagem || 'Erro ao inativar aluno.'
                );
              }

            } catch (erro) {

              console.log(erro);

              Alert.alert(
                'Erro',
                'Não foi possível conectar com o servidor.'
              );
            }
          },
        },
      ]
    );
  }

  // TELA LISTA DE ALUNOS
  // ======================================================

  if (telaAluno === 'lista') {

    return (

      <ScrollView
        contentContainerStyle={styles.container}
      >

        <Text style={styles.titulo}>
          Lista de Alunos
        </Text>

          <Text style={{ marginTop: 20 }}>
            Nenhum aluno cadastrado.
          </Text>

        ) : (

          alunos.map((aluno) => (

            <TouchableOpacity
  key={aluno.id}
  style={styles.cardAluno}
  onPress={() => prepararEdicao(aluno)}
>

  <Text style={styles.nomeAluno}>
    {aluno.nome}
  </Text>

  <Text>
    ID: {aluno.id}
  </Text>

  <Text>
    Status: {aluno.status === 'A' ? 'Ativo' : 'Inativo'}
  </Text>

  <Text
    style={{
      marginTop: 10,
      color: '#1976D2',
      fontWeight: 'bold',
    }}
  >
    Toque para visualizar
  </Text>

</TouchableOpacity>

          ))
        )}


        <TouchableOpacity
          style={styles.botao}
          onPress={consultarAlunos}
        >

          <Text style={styles.textoBotao}>
            Atualizar Lista
          </Text>

        </TouchableOpacity>


        <TouchableOpacity
          style={styles.botaoVoltar}
          onPress={() => setTelaAluno('inicio')}
        >

          <Text style={styles.textoBotao}>
            Voltar
          </Text>

        </TouchableOpacity>

      </ScrollView>
    );
  }
}

  // --- TELA SOBRE ---

    return (
      <View style={styles.container}>
        <Text style={styles.titulo}>Sobre o APP Scholar</Text>
        <Text style={styles.caixa}>

    );
  }
